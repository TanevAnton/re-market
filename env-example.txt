# Phone verification cascade: cheapest channel first, fall back on failure.
# Telegram ~$0.01, Viber ~EUR 0.017, SMS ~EUR 0.04. Twilio would be ~$0.20.
# With none of the credentials below set, PhoneVerifier falls back to the log
# channel - so signup works locally with no accounts and no spend. Look for
# "[verification] ... -> code 123456" in storage/logs/laravel.log, or run
# .\code.ps1 which digs out the latest one.
VERIFY_CHANNELS=telegram,viber,sms
VERIFY_CODE_TTL_MINUTES=10
VERIFY_MAX_ATTEMPTS=5
# Staging / LAN escape hatch. With APP_ENV=production the log channel refuses
# to run, because a channel that reports success while delivering nothing lets
# anyone who can read the log verify anyone's number. Set this to true ONLY on
# a test box with no real accounts.
VERIFY_ALLOW_LOG_CHANNEL=false

# Telegram Gateway: ~$0.01 per code, and FREE to your own phone number, which
# makes it the right channel to develop against. Get a token at
# https://gateway.telegram.org (log in with the Telegram account whose number
# you want to verify for free), then paste it here.
TELEGRAM_GATEWAY_TOKEN=

# Telegram BOT verification - free at any volume, no per-message cost.
# The user opens the bot, taps "share my number", and Telegram hands over the
# number it verified at signup. Create the bot with @BotFather and paste its
# token below - that is the only required value. Needs the listener running:
#   php artisan remarket:telegram-bot        (systemd unit in deploy/)
# Check the whole setup with:
#   php artisan remarket:telegram-doctor
